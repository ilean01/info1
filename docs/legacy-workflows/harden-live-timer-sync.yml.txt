name: INFO1 harden live timer sync

on:
  push:
    paths:
      - '.github/workflows/harden-live-timer-sync.yml'

permissions:
  contents: write

jobs:
  patch:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Patch bidirectional timer fallback and mobile auto update
        shell: bash
        run: |
          python3 - <<'PY'
          from pathlib import Path
          import re

          # ---- device-sync.js ----
          p = Path('device-sync.js')
          s = p.read_text(encoding='utf-8')

          old = '''  let timerPublishQueued = false;
  let timerPendingPublish = false;
'''
          new = '''  let timerPublishQueued = false;
  let timerPendingPublish = false;
  let timerStateFallbackBusy = false;
  let timerLastStateRevision = 0;
'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'timerStateFallbackBusy' not in s:
              raise SystemExit('timer variable target not found')

          old = '''      version: incomingVersion,
      source_id: row.source_id || null
    };
'''
          new = '''      version: incomingVersion,
      source_id: row.source_id || null,
      updated_at: row.updated_at || null
    };
'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'updated_at: row.updated_at || null' not in s:
              raise SystemExit('canonical timer row target not found')

          marker = '''  /* ===== ACTUALIZACIÓN DE APP ===== */
'''
          fallback = r'''  // Puente de compatibilidad: versiones viejas de la PWA pueden no escribir
  // info1_live_timer al tocar STOP, pero sí guardan el estado principal.
  // Comparamos fechas y, si info1_state es más nuevo que el canal live, lo
  // usamos como autoridad y reparamos inmediatamente el canal live.
  async function reconcileTimerFromFullState() {
    if (timerStateFallbackBusy || document.hidden) return false;
    const sb = window.INFO1_SUPABASE_CLIENT;
    const ctx = cloudContext();
    if (!sb || !ctx.connected || !ctx.workspaceId) return false;

    timerStateFallbackBusy = true;
    try {
      const result = await sb
        .from('info1_state')
        .select('state,revision,updated_at')
        .eq('workspace_id', ctx.workspaceId)
        .maybeSingle();
      if (result.error || !result.data) return false;

      const revision = Number(result.data.revision || 0);
      const stateUpdatedMs = Date.parse(result.data.updated_at || '') || 0;
      const liveUpdatedMs = Date.parse(timerCanonicalRow?.updated_at || '') || 0;

      if (revision <= timerLastStateRevision && stateUpdatedMs <= liveUpdatedMs) return false;
      timerLastStateRevision = Math.max(timerLastStateRevision, revision);

      // El canal live gana si fue actualizado después que el estado completo.
      if (liveUpdatedMs && stateUpdatedMs <= liveUpdatedMs) return false;

      const remoteState = result.data.state && typeof result.data.state === 'object'
        ? result.data.state
        : {};
      const remoteCrono = remoteState.__crono?.id && remoteState.__crono?.inicio
        ? remoteState.__crono
        : null;
      const remoteSnap = timerSnapshot(remoteCrono);
      const localSnap = timerSnapshot(currentCrono());
      if (remoteSnap === localSnap) return false;

      const synthetic = {
        active: !!remoteCrono,
        topic_id: remoteCrono?.id || null,
        started_at: remoteCrono?.inicio || null,
        version: timerLastVersion,
        updated_at: result.data.updated_at || new Date().toISOString(),
        source_id: 'full-state-fallback'
      };

      applySharedTimerRow(synthetic, true);

      // Reparar la tabla live para que el resto de los dispositivos converja.
      timerPendingPublish = true;
      await publishSharedTimer();
      return true;
    } catch (e) {
      console.warn('INFO1 timer full-state fallback:', e);
      return false;
    } finally {
      timerStateFallbackBusy = false;
    }
  }

'''
          if 'async function reconcileTimerFromFullState()' not in s:
              if marker not in s:
                  raise SystemExit('app update marker not found')
              s = s.replace(marker, fallback + marker, 1)

          old = '''      if (seen !== release) {
        appUpdateAvailable = true;
        pendingRelease = release;
        return true;
      }
'''
          new = '''      if (seen !== release) {
        appUpdateAvailable = true;
        pendingRelease = release;
        // Desde esta versión la PWA se actualiza sola. No recargamos encima
        // de una edición pendiente, pero un cronómetro sincronizado sí puede
        // sobrevivir a la recarga porque vive en info1_live_timer.
        if (!hasUnsyncedHumanChanges()) {
          setTimeout(() => reloadForNewAppVersion(true), 650);
        }
        return true;
      }
'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'reloadForNewAppVersion(true)' not in s:
              raise SystemExit('checkLatestApp target not found')

          old = '''  function reloadForNewAppVersion() {
    if (!appUpdateAvailable || hasUnsyncedHumanChanges() || hasActiveChrono()) return false;
    if (pendingRelease) localStorage.setItem(RELEASE_KEY, pendingRelease);
    const url = new URL(location.href);
    url.searchParams.set('_info1version', Date.now().toString());
    location.replace(url.toString());
    return true;
  }
'''
          new = '''  function reloadForNewAppVersion(force = false) {
    if (!appUpdateAvailable || hasUnsyncedHumanChanges()) return false;
    if (hasActiveChrono() && !force) return false;
    // El pagehide del HTML no debe interpretar esta recarga técnica como que
    // la persona salió de la ficha y detener el cronómetro.
    window.__INFO1_UPDATING_APP__ = true;
    if (pendingRelease) localStorage.setItem(RELEASE_KEY, pendingRelease);
    const url = new URL(location.href);
    url.searchParams.set('_info1version', Date.now().toString());
    location.replace(url.toString());
    return true;
  }
'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'window.__INFO1_UPDATING_APP__ = true;' not in s:
              raise SystemExit('reloadForNewAppVersion target not found')

          old = '''  setInterval(() => {
    setupSharedTimer();
    pollSharedTimer();
  }, 1500);

  setInterval(checkLatestApp, 45000);
'''
          new = '''  setInterval(() => {
    setupSharedTimer();
    pollSharedTimer();
  }, 1500);

  // Fallback para clientes móviles viejos: si STOP quedó sólo en info1_state,
  // la notebook lo detecta y repara el canal compartido.
  setInterval(reconcileTimerFromFullState, 1200);

  // Revisar la versión con más frecuencia; al volver a primer plano también se
  // ejecuta syncEverything(), por lo que el iPhone se pone al día solo.
  setInterval(checkLatestApp, 10000);
'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'setInterval(reconcileTimerFromFullState, 1200);' not in s:
              raise SystemExit('timer interval target not found')

          old = '''    pollSharedTimer,
    publishSharedTimer,
'''
          new = '''    pollSharedTimer,
    reconcileTimerFromFullState,
    publishSharedTimer,
'''
          if old in s:
              s = s.replace(old, new, 1)

          p.write_text(s, encoding='utf-8')

          # ---- index.html ----
          p = Path('index.html')
          s = p.read_text(encoding='utf-8')

          # Cache-bust the sync client on Safari/PWA.
          s = re.sub(r'(<script src="\./device-sync\.js\?v=)[^"]+("[^>]*></script>)',
                     r'\g<1>20261001-live4\g<2>', s, count=1)

          old = '''window.addEventListener("pagehide",()=>{
  const id=currentEdit?.id;
  if(id&&activeCrono()?.id===id)stopFichaTimer(id);
});'''
          new = '''window.addEventListener("pagehide",()=>{
  if(window.__INFO1_UPDATING_APP__)return;
  const id=currentEdit?.id;
  if(id&&activeCrono()?.id===id)stopFichaTimer(id);
});'''
          if old in s:
              s = s.replace(old, new, 1)
          elif 'if(window.__INFO1_UPDATING_APP__)return;' not in s:
              raise SystemExit('pagehide target not found')

          p.write_text(s, encoding='utf-8')
          PY

      - name: Commit patch
        shell: bash
        run: |
          git config user.name 'INFO1 sync bot'
          git config user.email 'actions@users.noreply.github.com'
          git add device-sync.js index.html
          if git diff --cached --quiet; then
            echo 'No changes'
            exit 0
          fi
          git commit -m 'fix: sync mobile stop via state fallback and auto update PWA'
          for attempt in 1 2 3 4 5; do
            git pull --rebase origin main && git push && exit 0
            sleep $((attempt * 2))
          done
          exit 1
